@Suite(.nativeIntegration, .serialized)
struct CursorSourceIntegrationTests {
  @Test(.timeLimit(.minutes(10))) func registeredVendorStop() async throws {
    let env = ProcessInfo.processInfo.environment
    _ = try #require(env["COMPUTER_MCP_CURSOR_ARCHIVE"])
    _ = try #require(env["COMPUTER_MCP_CURSOR_ARCHIVE_SHA256"])
    let source = URL(fileURLWithPath: try #require(env["COMPUTER_MCP_CURSOR_PLUGIN"]))
    let package = try PluginPackage.load(at: source)
    for (caller, provider) in [(1, true), (2, false)] {
      try await withCursor(package, caller: caller, provider: provider) { h in
        try await h.discover()
        let before = h.mcpObservations.members.count
        let hostBefore = h.base.observations.members.count
        let denied = try await h.call("cursor.acp.session.open")
        try cursorDenied(denied)
        try #require(
          denied.objectValue?["structuredContent"]?.objectValue?["error"]?.objectValue?["code"]
            == .string("permission.required"))
        try #require(h.mcpObservations.members.count == before)
        try #require(h.base.observations.members.count == hostBefore)
        print(
          "Cursor gate denial preserved (caller write: \(caller == 2), provider write: \(provider))"
        )
      }
    }

    var measured: [[String: Any]] = []
    var nativeStops: [[String: Any]] = []
    try await withCursor(package) { h in
      try await h.discover()
      let baseline = h.mcpObservations.members.count
      let hostBaseline = h.base.observations.members.count
      let a = try await h.open()
      let warm = h.mcpObservations.members.count
      let b = try await h.open()
      let aRoute = try await h.observeSession(a)
      let bRoute = try await h.observeSession(b)
      try #require(aRoute.connectionID == bRoute.connectionID)
      try #require(aRoute.instanceID == bRoute.instanceID)
      try #require(aRoute.acquiredBy != bRoute.acquiredBy)
      let peers = try h.vendorEffects()
      try #require(
        peers.compactMap { $0["phase"]?.stringValue } == ["probe", "workload", "probe", "workload"])
      let roots = try peers.map { try h.peerMember($0) }
      try #require(Set(roots).count == 4)
      let coldRoots = Array(roots.prefix(2))
      let warmRoots = Array(roots.suffix(2))
      let coldLaunches = Array(
        h.mcpObservations.members.dropFirst(baseline).prefix(warm - baseline))
      let adapterCandidates = Set(coldLaunches).subtracting(coldRoots)
      try #require(adapterCandidates.count == 1)
      let adapter = try #require(adapterCandidates.first)
      try h.requireMCPLaunches(coldLaunches, exactly: coldRoots + [adapter])
      try h.requireMCPLaunches(since: warm, exactly: warmRoots)
      measured = try h.verifyAcquisitions(first: coldRoots, second: warmRoots, adapter: adapter)
      print(
        "Cursor cold/warm proof counted five MCP roots and \(h.base.observations.members.count - hostBaseline) separate Host launches"
      )
      let aRoot = roots[1]
      let bRoot = roots[3]
      let aPID = aRoot.processID
      let bPID = bRoot.processID
      let aOwner = try h.processOwner(aRoot)
      let bOwner = try h.processOwner(bRoot)
      print("Cursor gate acquired A \(aPID) and B \(bPID) in adapter \(adapter.processID)")
      let noFD = try cursorValue(try await h.prompt(a, "env"))
      try #require(noFD["text"] == .string("[]"))
      let outside = String(
        decoding: try JSONSerialization.data(withJSONObject: h.outside.map(\.path)),
        as: UTF8.self)
      _ = try cursorValue(try await h.prompt(a, "boundaries:" + outside))
      let allowed = h.worktree.appendingPathComponent("cursor-selected-write")
      try #require(try String(contentsOf: allowed, encoding: .utf8) == "written")
      for url in h.outside {
        try #require(try String(contentsOf: url, encoding: .utf8) == "sentinel")
      }

      async let normal = h.prompt(a, "slow")
      try await h.waitVendorPrompt(pid: aPID, prompt: "slow")
      let normalStop = try await h.stop(a)
      try #require(try cursorValue(normalStop)["cancel_requested"] == .bool(true))
      let normalReply = try await normal
      h.reportStop(pid: aPID, prompt: "slow", stop: normalStop, reply: normalReply)
      let stopped = try cursorNativeValue(normalReply)
      try #require(stopped["stop_reason"] == .string("cancelled"))
      try #require(
        stopped["native_result"]?.objectValue?["usage"]?.objectValue?["tokens"] == .integer(3))
      try #require(stopped["closed"] == .bool(false))
      try #require(try await aOwner.members().contains(aRoot))
      let afterStop = try cursorValue(
        try await h.call("cursor.acp.requests.list", ["session": .string(a)]))
      try #require(afterStop["requests"] == .array([]))
      let resumed = try cursorValue(try await h.prompt(a, "hello after cancellation"))
      try #require(
        resumed["session"] == .string(a) && resumed["stop_reason"] == .string("end_turn"))
      try await h.waitVendorPrompt(pid: aPID, prompt: "hello after cancellation")
      try #require(try await h.observeSession(a) == aRoute)
      try #require(try await h.observeSession(b) == bRoute)
      try h.requireMCPLaunches(since: baseline, exactly: roots + [adapter])

      async let interactive = h.prompt(a, "permission-on-cancel")
      _ = try await h.waitPending(a)
      let beforeAnswer = try h.vendorEffects().count
      let interactiveStop = try await h.stop(a)
      try #require(try cursorValue(interactiveStop)["cancel_requested"] == .bool(true))
      let interactiveReply = try await interactive
      h.reportStop(
        pid: aPID, prompt: "permission-on-cancel", stop: interactiveStop, reply: interactiveReply)
      let interactiveResult = try cursorNativeValue(interactiveReply)
      try #require(interactiveResult["stop_reason"] == .string("cancelled"))
      let answers = try h.vendorEffects().dropFirst(beforeAnswer).filter {
        $0["phase"] == .string("answer") && $0["pid"]?.intValue == Int(aPID)
      }
      try #require(
        answers.compactMap { $0["response"] } == [
          .object(["outcome": .object(["outcome": .string("cancelled")])])
        ])
      let pendingAfterStop = try cursorValue(
        try await h.call("cursor.acp.requests.list", ["session": .string(a)]))
      try #require(pendingAfterStop["requests"] == .array([]))
      _ = try cursorValue(try await h.prompt(a, "hello after pending cancellation"))

      async let nativeError = h.prompt(a, "error-on-cancel")
      try await h.waitVendorPrompt(pid: aPID, prompt: "error-on-cancel")
      let errorStop = try await h.stop(a)
      _ = try cursorValue(errorStop)
      let errorReply = try await nativeError
      h.reportStop(pid: aPID, prompt: "error-on-cancel", stop: errorStop, reply: errorReply)
      let error = try cursorNativeValue(errorReply)["error"]?.objectValue
      try #require(error?["code"] == .string("vendor_error"))
      try #require(error?["message"]?.stringValue?.contains("-32023") == true)

      let isolationStarted = ContinuousClock.now
      let bPrompt = "complete-on-cancel"
      let bRequestID = "cursor-vendor-isolation-b"
      // Retain B's original MCP result after dispatch so its wait cannot hold a Gateway worker.
      let startedB = try cursorValue(
        try await h.call(
          "cursor.acp.session.prompt", ["session": .string(b), "prompt": .string(bPrompt)],
          startRequestID: bRequestID))
      try #require(startedB["request_id"] == .string(bRequestID))
      try #require(startedB["state"] == .string("running"))
      let bDownstreamID = try #require(startedB["downstream_request_id"]?.stringValue)
      try await h.waitVendorPrompt(pid: bPID, prompt: bPrompt)
      let bStarted = try h.vendorTurn(pid: bPID, prompt: bPrompt)
      let bNativeID = try #require(bStarted.first?["acp_id"]?.intValue)
      h.reportIsolation(
        "B retained MCP request \(bDownstreamID) reached peer as ACP request \(bNativeID)",
        started: isolationStarted)
      async let ignored = h.observedPrompt(
        a, "ignore-cancel", pid: aPID, started: isolationStarted)
      try await h.waitVendorPrompt(pid: aPID, prompt: "ignore-cancel")
      h.reportIsolation("A prompt reached peer", started: isolationStarted)
      let ignoredStop = try await h.stop(a)
      h.reportIsolation("A stop returned", started: isolationStarted)
      if let stopRequestID = ignoredStop.objectValue?["structuredContent"]?.objectValue?[
        "gateway_execution"]?.objectValue?["request_id"]?.stringValue,
        let audit = try? h.base.database.auditEvents(requestID: stopRequestID)
      {
        let admitted = audit.first { $0.errorCode == "permission.admitted" }?.occurredAt
        h.reportIsolation(
          "A stop audit request \(stopRequestID) admitted_ms "
            + (admitted.map { String(Int64($0.timeIntervalSince1970 * 1_000)) } ?? "none"),
          started: isolationStarted)
      }
      if ignoredStop.objectValue?["isError"] == .bool(true) {
        h.reportStop(pid: aPID, prompt: "ignore-cancel", stop: ignoredStop, reply: .null)
      }
      try #require(try cursorValue(ignoredStop)["cancel_requested"] == .bool(true))
      let ignoredReply = try await ignored
      h.reportStop(pid: aPID, prompt: "ignore-cancel", stop: ignoredStop, reply: ignoredReply)
      let unsettled = try cursorNativeValue(ignoredReply)["error"]?.objectValue
      try #require(unsettled?["code"] == .string("cancellation_unsettled"))
      try #require(unsettled?["message"]?.stringValue?.contains("did not settle") == true)
      let aMembers = try await aOwner.members()
      let bMembers = try await bOwner.members()
      let siblingAlive = bMembers.contains { $0.processID == bPID }
      h.reportIsolation(
        "A settled; A members \(aMembers.map(\.processID)), B members \(bMembers.map(\.processID))",
        started: isolationStarted)
      if !siblingAlive {
        let adapterOwner = try? h.processOwner(adapter)
        let adapterMembers = try? await adapterOwner?.members()
        h.reportIsolation(
          "adapter members \(adapterMembers?.map(\.processID) ?? [])",
          started: isolationStarted)
        h.reportAcquisitionLifetimes(measured, started: isolationStarted)
      }
      try #require(aMembers.isEmpty)
      try #require(siblingAlive)
      let bPending = try h.vendorTurn(pid: bPID, prompt: bPrompt)
      try #require(bPending.map { $0["phase"]?.stringValue } == ["prompt"])
      try #require(bPending.first?["acp_id"]?.intValue == bNativeID)
      let pendingReceipt = try await h.requestReceipt(bRequestID)
      try #require(pendingReceipt["state"] == .string("running"))
      try #require(pendingReceipt["downstream_request_id"] == .string(bDownstreamID))

      let completionStop = try await h.stop(b)
      try #require(try cursorValue(completionStop)["cancel_requested"] == .bool(true))
      let completedReceipt = try await h.waitRequestReceipt(bRequestID)
      try #require(completedReceipt["state"] == .string("succeeded"), "\(completedReceipt)")
      try #require(completedReceipt["request_id"] == .string(bRequestID))
      try #require(completedReceipt["downstream_request_id"] == .string(bDownstreamID))
      let completionReply = try #require(completedReceipt["result"])
      h.reportStop(pid: bPID, prompt: bPrompt, stop: completionStop, reply: completionReply)
      let won = try cursorValue(completionReply)
      try #require(won["session"] == .string(b))
      try #require(won["stop_reason"] == .string("end_turn"))
      try #require(
        won["native_result"]?.objectValue?["usage"]?.objectValue?["tokens"] == .integer(7))
      let bSettled = try h.vendorTurn(pid: bPID, prompt: bPrompt)
      try #require(bSettled.map { $0["phase"]?.stringValue } == ["prompt", "cancel", "terminal"])
      try #require(bSettled.last?["acp_id"]?.intValue == bNativeID)
      try #require(bSettled.last?["outcome"]?.objectValue?["result"] == won["native_result"])
      try #require(try await h.observeSession(b) == bRoute)
      let freshB = try await h.verifySiblingIO(b, root: bRoot, suffix: "forced-stop")
      let freshTurn = try h.vendorTurn(pid: bPID, prompt: freshB)
      let freshNativeID = try #require(freshTurn.first?["acp_id"]?.intValue)
      try #require(freshNativeID > bNativeID)
      try #require(try h.nativeStarts == 2)
      try h.requireMCPLaunches(since: baseline, exactly: roots + [adapter])
      _ = try cursorValue(
        try await h.call("cursor.acp.session.close", ["session": .string(b)]))
      try #require(try await bOwner.members().isEmpty)

      // Correlate each addressed stop with the peer turn and then require that
      // these exact deliveries account for every cancel, including after close.
      let correlations = try [
        h.nativeStop(aRoot, prompt: "slow", stop: normalStop, promptReply: normalReply),
        h.nativeStop(
          aRoot, prompt: "permission-on-cancel", stop: interactiveStop,
          promptReply: interactiveReply, terminalLogged: false),
        h.nativeStop(
          aRoot, prompt: "error-on-cancel", stop: errorStop, promptReply: errorReply),
        h.nativeStop(
          aRoot, prompt: "ignore-cancel", stop: ignoredStop, promptReply: ignoredReply,
          terminalLogged: false),
        h.nativeStop(
          bRoot, prompt: bPrompt, stop: completionStop, promptReply: .object(completedReceipt)),
      ]
      let effects = try h.vendorEffects()
      try #require(effects.filter { $0["phase"] == .string("probe") }.count == 2)
      let actualCancels = effects.filter { $0["phase"] == .string("cancel") }
      try #require(actualCancels == correlations.map(\.event))
      nativeStops = correlations.map(\.receipt)
    }
    let evidence = URL(fileURLWithPath: try #require(env["COMPUTER_MCP_CURSOR_CASES"]))
    let cases = [
      "dual-authority", "cold-warm-ownership", "seatbelt-boundaries", "ordinary-cancel",
      "native-result-races", "pending-answer-cancel", "forced-stop-isolation", "cleanup",
      "startup-cancellation", "lost-start-reply", "pending-start-closure",
    ]
    var details: [String: Any] = ["acquisitions": measured, "native_stops": nativeStops]
    details["startup_cancellation"] = try await registeredStartupCancellation(package)
    details["lost_start_reply"] = try await registeredLostStartReply(package)
    details["pending_start_closure"] = try await registeredPendingStartClosure(package)
    details["kind"] = "deterministic"
    details["tests_executed"] = 1
    let runID = try #require(env["COMPUTER_MCP_CURSOR_RUN_ID"])
    details["run_id"] = runID
    details["cases"] = cases
    try JSONSerialization.data(withJSONObject: details).write(to: evidence, options: .atomic)
  }

  private func registeredStartupCancellation(_ package: PluginPackage) async throws
    -> [String: Any]
  {
    var evidence: [String: Any] = [:]
    for withdraw in [false, true] {
      try await withCursor(package, observeStartup: true) { h in
        try await h.discover()
        let b = try await h.open()
        let bRoot = try h.lastWorkload()
        let bPrompt = "complete-on-cancel"
        let bRequestID = "startup-sibling"
        let bStarted = try cursorValue(
          try await h.call(
            "cursor.acp.session.prompt", ["session": .string(b), "prompt": .string(bPrompt)],
            startRequestID: bRequestID))
        try #require(bStarted["state"] == .string("running"))
        let bDownstream = try #require(bStarted["downstream_request_id"])
        try await h.waitVendorPrompt(pid: bRoot.processID, prompt: bPrompt)
        let bNative = try #require(
          try h.vendorTurn(pid: bRoot.processID, prompt: bPrompt).first?["acp_id"])
        let requestID = withdraw ? "startup-withdraw" : "startup-control"
        let before = h.mcpObservations.members.count
        let startCount = try h.nativeStarts
        try h.armStartupFault("hold-launch")
        let started = try await h.startOpen(requestID)
        let downstream = try #require(started["downstream_request_id"]?.stringValue)
        let held = try await h.waitTransport("launch-held")
        let boundary = try await h.startupBoundary(held)
        let beforeStop = try h.startupChildren(boundary)
        let probeRoot = try #require(beforeStop.probe.recovery?.root)
        try #require(beforeStop.probe.completed && beforeStop.probe.recovery?.sandboxed == true)
        try #require(!beforeStop.workload.completed && beforeStop.workload.recovery == nil)
        try h.requireMCPLaunches(since: before, exactly: [probeRoot])
        try h.requirePeerEvidence(probeRoot, phase: "probe")

        var trial = h.startupEvidence(boundary, requestID: requestID, downstream: downstream)
        trial["probe_state"] = "retired"
        trial["probe_pid"] = Int(probeRoot.processID)
        if withdraw {
          let cancelled = try cursorValue(
            try await h.consumer.generations.callToolForMCPAsync(
              name: "mcp.requests.cancel",
              arguments: .object([
                "server": .string("cursor-fixture"), "request_id": .string(requestID),
              ]), context: .empty))
          try #require(cancelled["cancellation_requested"] == .bool(true))
          try #require(cancelled["request_id"] == .string(requestID))
        }
        try h.releaseTransport("launch-held")
        let delivered = try await h.waitTransport("launch-delivered")
        try #require(delivered.request == held.request)
        let terminal = try await h.waitRequestReceipt(requestID)
        try #require(terminal["request_id"] == .string(requestID))
        try #require(terminal["downstream_request_id"] == .string(downstream))
        let reply = try #require(terminal["result"])
        if withdraw {
          try #require(delivered.reply?["error"] == .string("cancelled"))
          try #require(terminal["state"] == .string("failed"), "\(terminal)")
          try #require(terminal["cancellation"] == .string("sent"))
          try #require(reply.objectValue?["isError"] == .bool(true))
          let payload = reply.objectValue?["structuredContent"]?.objectValue?["result"]
          try #require(payload?.objectValue?["error"]?.objectValue?["code"] == .string("cancelled"))
          try await h.requireNeverLaunched(boundary, probe: probeRoot)
          try h.requireMCPLaunches(since: before, exactly: [probeRoot])
          try #require(try h.nativeStarts == startCount)
          trial["workload_state"] = "never_dispatched"
          trial["workload_starts"] = 0
          trial["result"] = "cancelled"
          try #require(
            boundary.adapter != bRoot && bRoot != probeRoot && boundary.adapter != probeRoot)
          evidence["adapter_pid"] = Int(boundary.adapter.processID)
          evidence["sibling_pid"] = Int(bRoot.processID)
        } else {
          let hostReceipt = try h.startupReply(delivered, boundary: boundary)
          try #require(hostReceipt["retired"] == .bool(false))
          try #require(terminal["state"] == .string("succeeded"), "\(terminal)")
          let session = try #require(try cursorValue(reply)["session"]?.stringValue)
          _ = try await h.observeSession(session)
          let live = try h.startupChildren(boundary)
          let root = try #require(live.workload.recovery?.root)
          try #require(!live.workload.completed && live.workload.recovery?.sandboxed == true)
          try h.requireMCPLaunches(since: before, exactly: [probeRoot, root])
          try h.requirePeerEvidence(root, phase: "workload")
          try #require(try h.nativeStarts == startCount + 1)
          _ = try cursorValue(
            try await h.call("cursor.acp.session.close", ["session": .string(session)]))
          let owner = try h.processOwner(root)
          try #require(try await owner.members().isEmpty)
          trial["workload_pid"] = Int(root.processID)
          trial["workload_starts"] = 1
          trial["result"] = "opened"
        }
        let siblingPending = try h.vendorTurn(pid: bRoot.processID, prompt: bPrompt)
        try #require(siblingPending.map { $0["phase"] } == [.string("prompt")])
        let bStop = try cursorValue(try await h.stop(b))
        try #require(bStop["cancel_requested"] == .bool(true))
        let finished = try await h.waitRequestReceipt(bRequestID)
        try #require(finished["state"] == .string("succeeded"))
        try #require(finished["downstream_request_id"] == bDownstream)
        let bReply = try cursorValue(try #require(finished["result"]))
        try #require(
          bReply["session"] == .string(b) && bReply["stop_reason"] == .string("end_turn"))
        let siblingSettled = try h.vendorTurn(pid: bRoot.processID, prompt: bPrompt)
        try #require(
          siblingSettled.map { $0["phase"] } == [
            .string("prompt"), .string("cancel"), .string("terminal"),
          ])
        try #require(siblingSettled.last?["acp_id"] == bNative)
        try #require(
          siblingSettled.last?["outcome"]?.objectValue?["result"] == bReply["native_result"])
        _ = try await h.verifySiblingIO(
          b, root: bRoot, suffix: withdraw ? "after-startup-cancellation" : "after-control-close")
        _ = try cursorValue(try await h.call("cursor.acp.session.close", ["session": .string(b)]))
        let bOwner = try h.processOwner(bRoot)
        try #require(try await bOwner.members().isEmpty)
        evidence[withdraw ? "cancelled" : "control"] = trial
      }
    }
    return evidence
  }

  private func registeredLostStartReply(_ package: PluginPackage) async throws -> [String: Any] {
    var evidence: [String: Any] = [:]
    try await withCursor(package, observeStartup: true) { h in
      try await h.discover()
      let b = try await h.open()
      let bRoot = try h.lastWorkload()
      let before = h.mcpObservations.members.count
      try h.armStartupFault("lose-reply")
      let requestID = "startup-lost-reply"
      let started = try await h.startOpen(requestID)
      let downstream = try #require(started["downstream_request_id"]?.stringValue)
      let dropped = try await h.waitTransport("launch-dropped")
      let boundary = try await h.startupBoundary(dropped)
      let launch = try h.startupReply(dropped, boundary: boundary)
      try #require(launch["retired"] == .bool(false))
      let recovered = try await h.waitTransport("receipt", timeout: .seconds(14))
      try h.requireNewRequest(recovered, after: dropped)
      let lookup = try h.startupReply(recovered, boundary: boundary)
      try #require(
        lookup["probe_state"] == .string("retired") && lookup["probe_settled"] == .bool(true))
      try #require(
        lookup["workload_state"] == .string("pending") && lookup["workload_settled"] == .bool(false)
      )
      try #require(lookup["retired"] == .bool(false))
      let live = try h.startupChildren(boundary)
      let root = try #require(live.workload.recovery?.root)
      let probe = try #require(live.probe.recovery?.root)
      try #require(live.probe.completed && !live.workload.completed)
      try h.requirePeerEvidence(root, phase: "workload")
      try h.requirePeerEvidence(probe, phase: "probe")
      let durable = try h.phaseJournal(boundary.workloadID, boundary: boundary)
      try #require(durable.recovery.root == root && !durable.completed && !durable.released)
      let aOwner = try h.processOwner(root)
      try #require(try await aOwner.members().contains(root))
      let bOwner = try h.processOwner(bRoot)
      try #require(try await bOwner.members().contains(bRoot))
      _ = try await h.verifySiblingIO(b, root: bRoot, suffix: "while-launch-reply-lost")
      try h.releaseTransport("receipt")

      let retired = try await h.waitTransport("retire")
      try h.requireNewRequest(retired, after: recovered)
      let settled = try h.startupReply(retired, boundary: boundary)
      try #require(
        settled["probe_state"] == .string("retired") && settled["probe_settled"] == .bool(true))
      try #require(
        settled["workload_state"] == .string("retired")
          && settled["workload_settled"] == .bool(true))
      try #require(settled["retired"] == .bool(true))
      let complete = try h.startupChildren(boundary)
      try #require(complete.probe.completed && complete.probe.recovery?.root == probe)
      try #require(complete.workload.completed && complete.workload.recovery?.root == root)
      let journal = try h.phaseJournal(boundary.workloadID, boundary: boundary)
      try #require(journal.completed && journal.recovery.root == root)
      try #require(try await aOwner.members().isEmpty)
      try h.releaseTransport("retire")

      let terminal = try await h.waitRequestReceipt(requestID)
      try #require(terminal["state"] == .string("failed"), "\(terminal)")
      try #require(terminal["downstream_request_id"] == .string(downstream))
      let reply = try #require(terminal["result"])
      try #require(reply.objectValue?["isError"] == .bool(true))
      let payload = reply.objectValue?["structuredContent"]?.objectValue?["result"]
      try #require(payload?.objectValue?["error"]?.objectValue?["code"] == .string("timeout"))
      let repeated = try cursorValue(
        try await h.call(
          "cursor.acp.session.open", ["permission_policy": .string("manual")],
          startRequestID: requestID))
      try #require(repeated["state"] == .string("failed"))

  private func registeredPendingStartClosure(_ package: PluginPackage) async throws -> [String: Any]
  {
    var evidence: [String: Any] = [:]
    for dispatched in [false, true] {
      try await withCursor(package, observeStartup: true) { h in
        try await h.discover()
        _ = try await h.open()

        let siblingOwner = try h.processOwner(sibling)
        let before = h.mcpObservations.members.count
        try h.armStartupFault(dispatched ? "hold-launch-reply" : "hold-launch")
        let requestID =
          dispatched ? "startup-close-after-dispatch" : "startup-close-before-dispatch"
        let started = try await h.startOpen(requestID)
        let downstream = try #require(started["downstream_request_id"]?.stringValue)
        let held = try await h.waitTransport(dispatched ? "launch-reply-held" : "launch-held")
        let boundary = try await h.startupBoundary(held)
        let live = try h.startupChildren(boundary)
        let probe = try #require(live.probe.recovery?.root)
        let workload: ExecutionMember?
        try #require(live.probe.completed && !live.workload.completed)
        try h.requirePeerEvidence(probe, phase: "probe")
        var roots = [probe]
        if dispatched {
          let launch = try h.startupReply(held, boundary: boundary)
          try #require(
            launch["retired"] == .bool(false) && launch["workload_settled"] == .bool(false))
          let root = try await h.waitStartedWorkload(boundary)
          workload = root
          roots.append(root)
        } else {
          try #require(live.workload.recovery == nil)
          workload = nil
        }
        let owner = try await h.vendorRequestOwner(requestID)
        let clock = ContinuousClock()
        let start = clock.now
        let result = try await h.closeVendorConnection(owner)
        try #require(start.duration(to: clock.now) < .seconds(8))
        try #require(result["transport_closed"] == .bool(true))
        try #require(result["connection_id"] == .string(owner.connectionID.uuidString))
        try #require(result["managed_process_exit_confirmed"] == .bool(true))
        try #require(result["work_cleanup"] == .string("confirmed"), "\(result)")
        try #require(result["remaining_owners"] == .integer(0), "\(result)")
        if let workload {
          let paired = try h.startupChildren(boundary)
          try #require(paired.probe.completed && paired.probe.recovery?.root == probe)
          try #require(paired.workload.completed && paired.workload.recovery?.root == workload)
          for (child, root) in [(boundary.probeID, probe), (boundary.workloadID, workload)] {
            let journal = try h.phaseJournal(child, boundary: boundary)
            try #require(journal.completed && journal.recovery.root == root)
            let process = try h.processOwner(root)
            try #require(try await process.members().isEmpty)
          }
        } else {
          try await h.requireNeverLaunched(boundary, probe: probe)
          try h.requireNoTransportEvent("launch-delivered")
        }
        try h.requireNoTransportEvent("launch-replayed")
        try #require(try await siblingOwner.members().isEmpty)
        let adapter = try h.processOwner(boundary.adapter)
        try #require(try await adapter.members().isEmpty)
        try h.requireMCPLaunches(since: before, exactly: roots)
        let terminal = try await h.waitRequestReceipt(requestID)
        try #require(terminal["state"] == .string("outcome_unknown"))
        try #require(terminal["cleanup"] == .string("confirmed"))
        try #require(terminal["downstream_request_id"] == .string(downstream))
        try #require(try h.nativeStarts == 1)
        var trial = h.startupEvidence(boundary, requestID: requestID, downstream: downstream)
        trial["probe_state"] = "retired"
        trial["workload_starts"] = roots.count - 1
        trial["workload_state"] = dispatched ? "retired" : "never_dispatched"
        trial["result"] = "outcome_unknown"
        trial["cleanup"] = "confirmed"
        evidence[dispatched ? "after_dispatch" : "before_dispatch"] = trial
      }
    }
    return evidence
  }

private func cursorWorkingDirectory(_ reported: String, matches selected: URL) throws {
  let actual = URL(fileURLWithPath: reported, isDirectory: true)
  for location in [actual, selected] {
    var isDirectory: ObjCBool = false
    try #require(FileManager.default.fileExists(atPath: location.path, isDirectory: &isDirectory))
    try #require(isDirectory.boolValue)
  }
  try #require(
    actual.resolvingSymlinksInPath().standardizedFileURL
      == selected.resolvingSymlinksInPath().standardizedFileURL)
}

private func cursorValue(_ value: JSONValue) throws -> [String: JSONValue] {
  try #require(value.objectValue?["isError"] != .bool(true), "\(value)")
  var result = try #require(
    value.objectValue?["structuredContent"]?.objectValue?["result"]?.objectValue)
  if let nested = result["structuredContent"]?.objectValue?["result"]?.objectValue {
    try #require(result["isError"] != .bool(true), "\(value)")
    result = nested
  }
  return result
}
private func cursorNativeValue(_ value: JSONValue) throws -> [String: JSONValue] {
  let outer = try #require(
    value.objectValue?["structuredContent"]?.objectValue?["result"]?.objectValue)
  return try #require(outer["structuredContent"]?.objectValue?["result"]?.objectValue, "\(value)")
}

private func cursorDenied(_ value: JSONValue) throws {
  try #require(value.objectValue?["isError"] == .bool(true), "\(value)")
  let code = try #require(
    value.objectValue?["structuredContent"]?.objectValue?["error"]?.objectValue?["code"]?
      .stringValue)
  try #require(
    code != "permission.catalog_required", "Catalog discovery is a fixture precondition: \(value)")
}
private func withCursor(
  _ package: PluginPackage, caller: Int = 2, provider: Bool = true,
  use: Bool = true, observeStartup: Bool = false, body: (CursorHarness) async throws -> Void
) async throws {
  let h = try await CursorHarness.make(
    package, caller: caller, provider: provider, use: use, observeStartup: observeStartup)
  do { try await body(h) } catch {
    do { try await h.close() } catch { Issue.record("Cursor cleanup failed: \(error)") }
    throw error
  }
  try await h.close()
}

  func startupBoundary(_ launch: CursorTransportEvent) async throws -> CursorStartupBoundary {
    let bind = try await waitTransport("bind")
    let probe = try await waitTransport("probe")
    let binding = try #require(bind.result)
    let acquisition = try #require(
      launch.request["acquisition"]?.stringValue.flatMap { UUID(uuidString: $0) })
    let probeID = try #require(binding["probe"]?.stringValue.flatMap { UUID(uuidString: $0) })
    let workloadID = try #require(binding["workload"]?.stringValue.flatMap { UUID(uuidString: $0) })
    try #require(
      launch.request["op"] == .string("launch") && launch.request["child"] == binding["workload"])
    try #require(bind.request["op"] == .string("bind") && probe.request["op"] == .string("probe"))
    try #require(probe.request["child"] == binding["probe"])
    for event in [bind, probe, launch] {
      try #require(event.adapterPID == launch.adapterPID)
      try #require(event.request["acquisition"] == launch.request["acquisition"])
      try #require(event.request["call"] == launch.request["acquisition"])
    }
    let adapter = try processMember(pid: launch.adapterPID)
    try #require(try processOwner(adapter).recoveryState().sandboxed)
    let matching = try gateway.permissionRuntime.store.readWithDatabase { database, ledger in
      try ledger.executions.values.compactMap { execution -> UUID? in
        guard execution.delegatedTo == .package("cursor"),
          let record = try ExecutionRuntimeRecord.load(execution.id, database),
          record.children[probeID] != nil, record.children[workloadID] != nil
        else { return nil }
        try #require(record.children.count == 2)
        return execution.id
      }
    }
    try #require(matching.count == 1)
    let execution = try #require(matching.first)
    let boundary = CursorStartupBoundary(
      acquisitionID: acquisition, executionID: execution, probeID: probeID, workloadID: workloadID,
      adapter: adapter)
    let receipt = try startupReply(probe, boundary: boundary)
    try #require(
      receipt["probe_state"] == .string("retired") && receipt["probe_settled"] == .bool(true))
    try #require(receipt["workload_settled"] == .bool(false))
    return boundary
  }

  func observedPrompt(
    _ session: String, _ text: String, pid: Int32, started: ContinuousClock.Instant
  ) async throws -> JSONValue {
    reportIsolation("starting \(text) on \(pid)", started: started)
    do {
      let reply = try await prompt(session, text)
      let outer = reply.objectValue?["structuredContent"]?.objectValue
      let inner = outer?["result"]?.objectValue
      let result = inner?["structuredContent"]?.objectValue?["result"]?.objectValue
      let requestID = outer?["gateway_execution"]?.objectValue?["request_id"]?.stringValue
      let audit = requestID.flatMap { try? base.database.auditEvent(requestID: $0) }
      let outcome = JSONValue.object([
        "request_id": requestID.map(JSONValue.string) ?? .null,
        "gateway_duration_ms": audit?.durationMilliseconds.map { .integer(Int64($0)) } ?? .null,
        "outer_error": outer?["error"] ?? .null,
        "adapter_error": result?["error"] ?? .null,
        "native_result": result?["native_result"] ?? .null,
      ])
      let encoder = JSONEncoder()
      encoder.outputFormatting = [.sortedKeys]
      let data = try encoder.encode(outcome)
      reportIsolation(
        "\(text) returned " + String(decoding: data.prefix(2_048), as: UTF8.self),
        started: started)
      return reply
    } catch {
      reportIsolation("\(text) threw \(error)", started: started)
      throw error
    }
  }
  func reportIsolation(_ event: String, started: ContinuousClock.Instant) {
    let elapsed = started.duration(to: .now).components
    let elapsedMS = elapsed.seconds * 1_000 + elapsed.attoseconds / 1_000_000_000_000_000
    let wallMS = Int64(Date().timeIntervalSince1970 * 1_000)
    print("Cursor isolation [\(wallMS) +\(elapsedMS)ms] \(event)")
  }
  func reportAcquisitionLifetimes(
    _ acquisitions: [[String: Any]], started: ContinuousClock.Instant
  ) {
    let ids = acquisitions.compactMap {
      ($0["execution"] as? String).flatMap(UUID.init(uuidString:))
    }
    do {
      let states = try gateway.permissionRuntime.store.readWithDatabase { database, ledger in
        try ids.map { id -> String in
          guard let execution = ledger.executions[id] else { return "\(id) unavailable" }
          let record = try ExecutionRuntimeRecord.load(id, database)
          let ended =
            execution.endedAt.map {
              String(Int64($0.timeIntervalSince1970 * 1_000))
            } ?? "none"
          let children =
            record?.children.values.compactMap { child -> String? in
              guard let pid = child.recovery?.root.processID else { return nil }
              return "\(pid) completed=\(child.completed)"
            }.sorted() ?? []
          let limit = execution.limitEndReason?.rawValue ?? "none"
          return
            "\(id) running=\(execution.isRunning) end_ms=\(ended) limit_end=\(limit) children=\(children)"
        }
      }
      for state in states { reportIsolation(state, started: started) }
    } catch {
      reportIsolation("ledger read failed \(error)", started: started)
    }
  }

  func nativeStop(
    _ root: ExecutionMember, prompt: String, stop: JSONValue, promptReply: JSONValue,
    terminalLogged: Bool = true
  ) throws -> (event: [String: JSONValue], receipt: [String: Any]) {
    let turn = try vendorTurn(pid: root.processID, prompt: prompt)
    let expectedPhases: [String?] =
      terminalLogged ? ["prompt", "cancel", "terminal"] : ["prompt", "cancel"]
    try #require(turn.map { $0["phase"]?.stringValue } == expectedPhases, "\(turn)")
    let event = try #require(turn.dropFirst().first)
    let acpID = try #require(event["acp_id"]?.intValue)
    try #require(event["native_session"] == turn.first?["native_session"])
    let stopID = try #require(
      stop.objectValue?["structuredContent"]?.objectValue?["gateway_execution"]?.objectValue?[
        "request_id"]?.stringValue)
    let promptID = try #require(
      promptReply.objectValue?["request_id"]?.stringValue
        ?? promptReply.objectValue?["structuredContent"]?.objectValue?["gateway_execution"]?
        .objectValue?["request_id"]?.stringValue)
    var receipt: [String: Any] = [
      "prompt": prompt, "peer_pid": Int(root.processID), "peer_identity": String(root.identity),
      "acp_id": acpID, "stop_request_id": stopID, "prompt_request_id": promptID,
    ]
    if let downstream = promptReply.objectValue?["downstream_request_id"]?.stringValue {
      receipt["downstream_request_id"] = downstream
    }
    return (event, receipt)
  }

  func verifyAcquisitions(
    first: [ExecutionMember], second: [ExecutionMember], adapter: ExecutionMember
  ) throws -> [[String: Any]] {
    try gateway.permissionRuntime.store.readWithDatabase { database, ledger in
      var rows: [[String: Any]] = []
      var executions: Set<UUID> = []
      var proofs: Set<UUID> = []
      var children: Set<UUID> = []
      for pair in [first, second] {
        let roots = Set(pair)
        var matched = false
        for execution in ledger.executions.values where execution.delegatedTo == .package("cursor")
        {
          guard let record = try ExecutionRuntimeRecord.load(execution.id, database) else {
            continue
          }
          let slots = record.children.filter { _, record in
            record.recovery.map { roots.contains($0.root) } ?? false
          }
          guard slots.count == 2 else { continue }
          let newExecution = executions.insert(execution.id).inserted
          try #require(!matched && newExecution)
          matched = true
          try #require(execution.authorizationIDs.count == 2)
          for id in execution.authorizationIDs {
            let newProof = proofs.insert(id).inserted
            try #require(newProof && ledger.authorizations[id]?.executionID == execution.id)
          }
          for id in slots.keys {
            let newChild = children.insert(id).inserted
            try #require(newChild)
          }
          var row: [String: Any] = [
            "execution": execution.id.uuidString,
            "authorization_ids": execution.authorizationIDs.map(\.uuidString).sorted(),
            "adapter_pid": adapter.processID,
            "adapter_identity": String(adapter.identity),
          ]
          if rows.isEmpty {
            try #require(record.children.count == 3)
            let adapterSlot = try #require(
              record.children.first { $0.value.recovery?.root == adapter })
            let inserted = children.insert(adapterSlot.key).inserted
            try #require(inserted)
            try verifyDurableLaunch(adapterSlot.key, in: record, root: adapter, completed: false)
            row["adapter_child"] = adapterSlot.key.uuidString
          } else {
            try #require(record.children.count == 2)
            row["adapter_child"] = rows[0]["adapter_child"]
          }
          for (phase, root) in zip(["probe", "workload"], pair) {
            let slot = try #require(slots.first { $0.value.recovery?.root == root })
            try verifyDurableLaunch(slot.key, in: record, root: root, completed: phase == "probe")
            row[phase + "_child"] = slot.key.uuidString
            row[phase + "_pid"] = root.processID
            row[phase + "_identity"] = String(root.identity)
          }
          rows.append(row)
        }
        try #require(matched)
      }
      return rows
    }
  }
  private func verifyDurableLaunch(
    _ child: UUID, in record: ExecutionRuntimeRecord, root: ExecutionMember, completed: Bool
  ) throws {
    let slot = try #require(record.children[child])
    try #require(slot.recovery?.root == root && slot.recovery?.sandboxed == true)
    try #require(slot.completed == completed)
    let entry = try phaseJournal(child, execution: record.id)
    try #require(entry.owner == record.ownerID && entry.boot == record.boot)
    try #require(entry.recovery.root == root && entry.completed == completed)
    _ = try processOwner(root)
  }
