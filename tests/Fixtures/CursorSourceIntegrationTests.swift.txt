    let env = ProcessInfo.processInfo.environment
    _ = try #require(env["COMPUTER_MCP_CURSOR_ARCHIVE"])
    _ = try #require(env["COMPUTER_MCP_CURSOR_ARCHIVE_SHA256"])
    let source = URL(fileURLWithPath: try #require(env["COMPUTER_MCP_CURSOR_PLUGIN"]))
    let package = try PluginPackage.load(at: source)
    for (caller, provider) in [(1, true), (2, false)] {
      try await withCursor(package, caller: caller, provider: provider) { h in

        let before = h.base.observations.launches.count
        let denied = try await h.call("cursor.acp.session.open")

    try await withCursor(package) { h in
      func providerLaunches() -> [ExecutionObservations.Launch] {
        h.base.observations.launches.filter { $0.executable != h.base.platform.gitExecutable }
      }
      try await h.discover()
      let cold = providerLaunches().count
      let a = try await h.open()
      let warm = providerLaunches().count
      let b = try await h.open()
      let launches = providerLaunches()

      let second = Array(launches.dropFirst(warm))
      try #require(first.count == 3, "\(first)")
      try #require(second.count == 2, "\(second)")

      try #require(adapter.arguments.last == "--host-vendor-execution")
      try #require(first.dropFirst().map(\.arguments) == [["--version"], ["acp"]])
      try #require(second.map(\.arguments) == [["--version"], ["acp"]])

      try #require(allLaunchesSandboxed)
      measured = try h.verifyAcquisitions(
        first: Array(first.dropFirst()), second: second, adapterPID: adapter.member.processID)
      let aPID = try #require(first.last?.member.processID)
      let bPID = try #require(second.last?.member.processID)
      print("Cursor gate acquired A \(aPID) and B \(bPID) in adapter \(adapter.member.processID)")
      print(
        "Cursor gate binaries \(adapter.executable) / \(first[1].executable) / \(second[0].executable)"
      )
      let aOwner = try #require(
        h.base.observations.ownerships.first {
          (try? $0.recoveryState().root.processID) == aPID

      let bOwner = try #require(
        h.base.observations.ownerships.first {
          (try? $0.recoveryState().root.processID) == bPID
        })
      let noFD = try cursorValue(try await h.prompt(a, "env"))
      try #require(noFD["text"] == .string("[]"))

        as: UTF8.self)
      _ = try cursorValue(try await h.prompt(a, "boundaries:" + outside))
      let allowed = h.worktree.appendingPathComponent("cursor-selected-write")
      try #require(try String(contentsOf: allowed, encoding: .utf8) == "written")
      for url in h.outside {
        try #require(try String(contentsOf: url, encoding: .utf8) == "sentinel")
      }

      try await h.waitVendorPrompt(pid: aPID, prompt: "slow")

      try #require(try cursorValue(normalStop)["cancel_requested"] == .bool(true))
      let normalReply = try await normal
      h.reportStop(pid: aPID, prompt: "slow", stop: normalStop, reply: normalReply)
      let stopped = try cursorNativeValue(normalReply)
      try #require(stopped["stop_reason"] == .string("cancelled"))
      try #require(
        stopped["native_result"]?.objectValue?["usage"]?.objectValue?["tokens"] == .integer(3))
      try #require(try await aOwner.members().contains(where: { $0.processID == aPID }))
      _ = try cursorValue(try await h.prompt(a, "hello after cancellation"))

      try #require(interactiveStop["cancel_requested"] == .bool(true))
      let interactiveResult = try cursorNativeValue(try await interactive)
      try #require(interactiveResult["stop_reason"] == .string("cancelled"))
      let answers = try h.vendorEffects().dropFirst(beforeAnswer).filter {
        $0["phase"] == .string("answer") && $0["pid"]?.intValue == Int(aPID)

        answers.compactMap { $0["response"] } == [
          .object(["outcome": .object(["outcome": .string("cancelled")])])

      try #require(error?["message"]?.stringValue?.contains("-32023") == true)

        try await h.call(
          "cursor.acp.session.prompt", ["session": .string(b), "prompt": .string(bPrompt)],

      try #require(startedB["state"] == .string("running"))

        "B retained MCP request \(bDownstreamID) reached peer as ACP request \(bNativeID)",
        started: isolationStarted)

        a, "ignore-cancel", pid: aPID, started: isolationStarted)
      try await h.waitVendorPrompt(pid: aPID, prompt: "ignore-cancel")
      h.reportIsolation("A prompt reached peer", started: isolationStarted)
      let ignoredStop = try await h.stop(a)
      h.reportIsolation("A stop returned", started: isolationStarted)

        "gateway_execution"]?.objectValue?["request_id"]?.stringValue,

      try #require(bPending.first?["acp_id"]?.intValue == bNativeID)

      try #require(freshNativeID > bNativeID)
      let effects = try h.vendorEffects()
      try #require(effects.filter { $0["phase"]?.stringValue == "probe" }.count == 2)
      try #require(
        effects.filter {
          $0["pid"]?.intValue == Int(aPID) && $0["phase"]?.stringValue == "cancel"

        effects.filter {

      try #require(providerLaunches().count == launches.count)
      _ = try cursorValue(
        try await h.call("cursor.acp.session.close", ["session": .string(b)]))
      try #require(try await bOwner.members().isEmpty)
    }
    let evidence = URL(fileURLWithPath: try #require(env["COMPUTER_MCP_CURSOR_CASES"]))
    var cases = [
      "dual-authority", "cold-warm-ownership", "seatbelt-boundaries", "ordinary-cancel",

  #if DEBUG
    private func registeredStartupCancellation(_ package: PluginPackage) async throws
      -> [String: Any]
    {
      var evidence: [String: Any] = [:]
      try await withCursor(package) { h in
        try await h.discover()
        // Discovery and delegated execution use separate pool authority.
        let executionBaseline = h.vendorLaunches.count
        let b = try await h.open()
        let launches = Array(h.vendorLaunches.dropFirst(executionBaseline))
        try #require(launches.count == 3)
        try #require(launches.dropFirst().map(\.arguments) == [["--version"], ["acp"]])
        let bPID = try #require(launches.last?.member.processID)
        let adapter = try #require(launches.first)
        try #require(adapter.arguments.last == "--host-vendor-execution")
        let adapterPID = adapter.member.processID
        let bOwner = try #require(
          h.base.observations.ownerships.first {
            (try? $0.recoveryState().root.processID) == bPID
          })
        let bPrompt = "complete-on-cancel"
        let bRequest = "startup-sibling"
        let bStarted = try cursorValue(
          try await h.call(
            "cursor.acp.session.prompt", ["session": .string(b), "prompt": .string(bPrompt)],
            startRequestID: bRequest))
        try #require(bStarted["state"] == .string("running"))
        let bDownstream = try #require(bStarted["downstream_request_id"])
        try await h.waitVendorPrompt(pid: bPID, prompt: bPrompt)
        let bPending = try h.vendorTurn(pid: bPID, prompt: bPrompt)
        try #require(bPending.map { $0["phase"]?.stringValue } == ["prompt"])
        let bNative = try #require(bPending.first?["acp_id"])

        for withdraw in [false, true] {
          let requestID = withdraw ? "startup-withdraw" : "startup-control"
          let gate = CursorStartupBarrier<CursorStartupBoundary>()
          let launchCount = h.vendorLaunches.count
          let startCount = try h.nativeStarts
          try await h.withStartupBarrier(gate) {
            let started = try cursorValue(
              try await h.call(
                "cursor.acp.session.open", ["permission_policy": .string("manual")],
                startRequestID: requestID))
            try #require(started["request_id"] == .string(requestID))
            try #require(started["state"] == .string("running"))
            let downstream = try #require(started["downstream_request_id"]?.stringValue)
            let boundary = try await gate.wait()
            let initialReceipts = boundary.receipts()
            try #require(initialReceipts.probe == "retired")
            try #require(initialReceipts.workload == "pending")
            let before = try h.startupChildren(boundary)
            let probePID = try #require(before.probe.recovery?.root.processID)
            try #require(before.probe.completed)
            try #require(before.workload.recovery == nil && !before.workload.completed)
            let atBoundary = Array(h.vendorLaunches.dropFirst(launchCount))
            try #require(atBoundary.map(\.arguments) == [["--version"]])
            try #require(atBoundary.first?.member.processID == probePID)
            try #require(atBoundary.first?.sandboxed == true)
            var trial: [String: Any] = [
              "acquisition": boundary.acquisitionID.uuidString,
              "execution": boundary.executionID.uuidString,
              "probe": boundary.probeID.uuidString,
              "workload": boundary.workloadID.uuidString,
              "probe_pid": Int(probePID),
              "probe_state": initialReceipts.probe,
              "request_id": requestID,
              "downstream_request_id": downstream,
            ]

            if withdraw {
              let cancellation = try cursorValue(
                try await h.gateway.callToolForMCPAsync(
                  name: "mcp.requests.cancel",
                  arguments: .object([
                    "server": .string("cursor-fixture"), "request_id": .string(requestID),
                  ]), context: .empty))
              try #require(cancellation["cancellation_requested"] == .bool(true))
              try #require(cancellation["request_id"] == .string(requestID))
              let sealed = try h.startupChildren(boundary)
              let receipts = boundary.receipts()
              try #require(receipts.probe == "retired")
              try #require(receipts.workload == "never_dispatched")
              try #require(sealed.probe.completed)
              try #require(sealed.probe.recovery?.root.processID == probePID)
              try #require(sealed.workload.completed && sealed.workload.recovery == nil)
              try #require(h.vendorLaunches.count == launchCount + 1)
              trial["workload_state"] = receipts.workload
              print("Cursor startup sealed \(boundary.acquisitionID) before workload bootstrap")
            }

            gate.release()
            let receipt = try await h.waitRequestReceipt(requestID)
            try #require(receipt["request_id"] == .string(requestID))
            try #require(receipt["downstream_request_id"] == .string(downstream))
            let reply = try #require(receipt["result"])
            if withdraw {
              try #require(receipt["state"] == .string("failed"), "\(receipt)")
              try #require(receipt["cancellation"] == .string("sent"))
              try #require(reply.objectValue?["isError"] == .bool(true))
              let payload = reply.objectValue?["structuredContent"]?.objectValue?["result"]
              try #require(
                payload?.objectValue?["error"]?.objectValue?["code"] == .string("cancelled"))
              try #require(h.vendorLaunches.count == launchCount + 1)
              try #require(try h.nativeStarts == startCount)
              trial["workload_starts"] = 0
              trial["result"] = "cancelled"
            } else {
              try #require(receipt["state"] == .string("succeeded"), "\(receipt)")
              let opened = try cursorValue(reply)
              let session = try #require(opened["session"]?.stringValue)
              let live = try h.startupChildren(boundary)
              let workloadPID = try #require(live.workload.recovery?.root.processID)
              try #require(!live.workload.completed)
              let launches = Array(h.vendorLaunches.dropFirst(launchCount))
              try #require(launches.map(\.arguments) == [["--version"], ["acp"]])
              try #require(launches.last?.member.processID == workloadPID)
              try #require(launches.last?.sandboxed == true)
              try #require(try h.nativeStarts == startCount + 1)
              let owner = try #require(
                h.base.observations.ownerships.first {
                  (try? $0.recoveryState().root.processID) == workloadPID
                })
              _ = try cursorValue(
                try await h.call("cursor.acp.session.close", ["session": .string(session)]))
              try #require(try await owner.members().isEmpty)
              trial["workload_pid"] = Int(workloadPID)
              trial["workload_starts"] = 1
              trial["result"] = "opened"
              print("Cursor startup control \(boundary.acquisitionID) launched \(workloadPID)")
            }
            evidence[withdraw ? "cancelled" : "control"] = trial
          }
        }

        try #require(try await bOwner.members().contains(where: { $0.processID == bPID }))
        let stillPending = try h.vendorTurn(pid: bPID, prompt: bPrompt)
        try #require(stillPending.map { $0["phase"]?.stringValue } == ["prompt"])
        try #require(stillPending.first?["acp_id"] == bNative)
        let stop = try cursorValue(try await h.stop(b))
        try #require(stop["cancel_requested"] == .bool(true))
        let completed = try await h.waitRequestReceipt(bRequest)
        try #require(completed["state"] == .string("succeeded"), "\(completed)")
        try #require(completed["downstream_request_id"] == bDownstream)
        let result = try cursorValue(try #require(completed["result"]))
        try #require(
          result["session"] == .string(b) && result["stop_reason"] == .string("end_turn"))
        let settled = try h.vendorTurn(pid: bPID, prompt: bPrompt)
        try #require(settled.map { $0["phase"]?.stringValue } == ["prompt", "cancel", "terminal"])
        try #require(settled.last?["acp_id"] == bNative)
        try #require(settled.last?["outcome"]?.objectValue?["result"] == result["native_result"])
        let fresh = "fresh B after startup cancellation"
        let reply = try cursorValue(try await h.prompt(b, fresh))
        try #require(reply["session"] == .string(b) && reply["stop_reason"] == .string("end_turn"))
        try await h.waitVendorPrompt(pid: bPID, prompt: fresh)
        let freshTurn = try h.vendorTurn(pid: bPID, prompt: fresh)
        let freshID = try #require(freshTurn.first?["acp_id"]?.intValue)
        let originalID = try #require(bNative.intValue)
        try #require(freshID > originalID)
        _ = try cursorValue(try await h.call("cursor.acp.session.close", ["session": .string(b)]))
        try #require(try await bOwner.members().isEmpty)
        let adapters = h.vendorLaunches.dropFirst(executionBaseline).filter {
          $0.arguments.last == "--host-vendor-execution"
        }
        try #require(adapters.map(\.member.processID) == [adapterPID])
        evidence["sibling_pid"] = Int(bPID)
        evidence["adapter_pid"] = Int(adapterPID)
      }
      return evidence
    }

    private func registeredLostStartReply(_ package: PluginPackage) async throws -> [String: Any] {
      var evidence: [String: Any] = [:]
      try await withCursor(package) { h in
        try await h.discover()
        let b = try await h.open()
        let bPID = try #require(h.vendorLaunches.last?.member.processID)
        let before = h.vendorLaunches.count
        let launch = CursorStartupBarrier<CursorStartupReply>()
        let lookup = CursorStartupBarrier<CursorStartupReply>()
        let retire = CursorStartupBarrier<CursorStartupReply>()
        try await h.withVendorReplyObserver(
          { reply in
            switch reply.operation {
            case .launch:
              await launch.park(reply)
              return true
            case .receipt:
              await lookup.park(reply)
              return false
            case .retire:
              await retire.park(reply)
              return false
            }
          },
          release: {
            launch.release()
            lookup.release()
            retire.release()
          }
        ) {
          let requestID = "startup-lost-reply"
          let arguments: [String: JSONValue] = ["permission_policy": .string("manual")]
          let started = try cursorValue(
            try await h.call("cursor.acp.session.open", arguments, startRequestID: requestID))
          try #require(started["state"] == .string("running"))
          try #require(started["request_id"] == .string(requestID))
          let downstream = try #require(started["downstream_request_id"]?.stringValue)
          let withheld = try await launch.wait()
          let boundary = withheld.boundary
          let probeID = JSONValue.string(boundary.probeID.uuidString.lowercased())
          let workloadID = JSONValue.string(boundary.workloadID.uuidString.lowercased())
          let pid = try await h.waitForNewWorkload(since: before)
          let live = try h.startupChildren(boundary)
          let member = try #require(live.workload.recovery?.root)
          try #require(member.processID == pid)
          try #require(withheld.requestedWorkloadID == boundary.workloadID)
          let durable = try h.workloadJournal(boundary)
          try #require(durable.recovery.root == member && !durable.completed)
          try #require(withheld.charged)
          print("Cursor startup reply dropped for \(boundary.workloadID) after peer \(pid) started")
          launch.release()

          let recovered = try await lookup.wait(timeout: .seconds(15))
          try #require(recovered.boundary.acquisitionID == boundary.acquisitionID)
          try #require(recovered.boundary.executionID == boundary.executionID)
          try #require(recovered.requestedWorkloadID == boundary.workloadID)
          try #require(recovered.payload["probe"] == probeID)
          try #require(recovered.payload["workload"] == workloadID)
          try #require(recovered.payload["probe_state"] == .string("retired"))
          try #require(recovered.payload["workload_state"] == .string("pending"))
          try #require(recovered.payload["retired"] == .bool(false))
          try #require(recovered.charged)

private func cursorWorkingDirectory(_ reported: String, matches selected: URL) throws {
  let actual = URL(fileURLWithPath: reported, isDirectory: true)
  for location in [actual, selected] {
    var isDirectory: ObjCBool = false
    try #require(FileManager.default.fileExists(atPath: location.path, isDirectory: &isDirectory))
    try #require(isDirectory.boolValue)
  }
  try #require(
    actual.resolvingSymlinksInPath().standardizedFileURL
      == selected.resolvingSymlinksInPath().standardizedFileURL)
}

private func cursorValue(_ value: JSONValue) throws -> [String: JSONValue] {
  try #require(value.objectValue?["isError"] != .bool(true), "\(value)")
  var result = try #require(
    value.objectValue?["structuredContent"]?.objectValue?["result"]?.objectValue)
  if let nested = result["structuredContent"]?.objectValue?["result"]?.objectValue {
    try #require(result["isError"] != .bool(true), "\(value)")
    result = nested
  }
  return result
}
private func cursorNativeValue(_ value: JSONValue) throws -> [String: JSONValue] {
  let outer = try #require(
    value.objectValue?["structuredContent"]?.objectValue?["result"]?.objectValue)
  return try #require(outer["structuredContent"]?.objectValue?["result"]?.objectValue, "\(value)")
}

private func cursorDenied(_ value: JSONValue) throws {
  try #require(value.objectValue?["isError"] == .bool(true), "\(value)")
  let code = try #require(
    value.objectValue?["structuredContent"]?.objectValue?["error"]?.objectValue?["code"]?
      .stringValue)
  try #require(
    code != "permission.catalog_required", "Catalog discovery is a fixture precondition: \(value)")
}
private func withCursor(
  _ package: PluginPackage, caller: Int = 2, provider: Bool = true,
  use: Bool = true, body: (CursorHarness) async throws -> Void
) async throws {
  let h = try await CursorHarness.make(package, caller: caller, provider: provider, use: use)
  do { try await body(h) } catch {
    do { try await h.close() } catch { Issue.record("Cursor cleanup failed: \(error)") }
    throw error
  }
  try await h.close()
}

      }

  func verifyAcquisitions(
    first: [ExecutionObservations.Launch], second: [ExecutionObservations.Launch], adapterPID: Int32
  ) throws -> [[String: Any]] {
    try gateway.permissionRuntime.store.readWithDatabase { database, ledger in
      var rows: [[String: Any]] = []
      var executions: Set<UUID> = []
      var proofs: Set<UUID> = []
      var children: Set<UUID> = []
      for pair in [first, second] {
        let pids = Set(pair.map { $0.member.processID })
        var matched = false
        for execution in ledger.executions.values where execution.delegatedTo == .package("cursor")
        {
          guard let record = try ExecutionRuntimeRecord.load(execution.id, database) else {
            continue
          }
          let slots = record.children.filter { _, record in
            record.recovery.map { pids.contains($0.root.processID) } ?? false
          }
          guard slots.count == 2 else { continue }
          let newExecution = executions.insert(execution.id).inserted
          try #require(!matched && newExecution)
          matched = true
          try #require(execution.authorizationIDs.count == 2)
          for id in execution.authorizationIDs {
            let newProof = proofs.insert(id).inserted
            try #require(newProof && ledger.authorizations[id]?.executionID == execution.id)
          }
          for id in slots.keys {
            let newChild = children.insert(id).inserted
            try #require(newChild)
          }
          if rows.isEmpty {
            try #require(
              record.children.values.contains { $0.recovery?.root.processID == adapterPID })
          } else {
            try #require(record.children.count == 2)
          }
          let phaseIDs = try pair.map { launch in
            try #require(slots.first(where: { $0.value.recovery?.root == launch.member })?.key)
          }
          rows.append([
            "execution": execution.id.uuidString,
            "authorization_ids": execution.authorizationIDs.map(\.uuidString).sorted(),
            "adapter_pid": adapterPID,
            "probe_pid": pair[0].member.processID,
            "probe_child": phaseIDs[0].uuidString,
            "workload_pid": pair[1].member.processID,
            "workload_child": phaseIDs[1].uuidString,
          ])
        }
        try #require(matched)
      }
      return rows
    }
  }
